# New Earth — form the hegemony after holding all seven empire titles.
# Works as soon as those titular titles exist on a character.

ge_form_new_earth_decision = {
	picture = { reference = "gfx/interface/illustrations/decisions/decision_misc.dds" }
	desc = ge_form_new_earth_decision_desc
	selection_tooltip = ge_form_new_earth_decision_tooltip
	major = yes
	ai_check_interval = 24

	is_shown = {
		is_landed = yes
		highest_held_title_tier >= tier_empire
		NOT = { has_title = title:e_ge_new_earth }
	}

	is_valid = {
		has_title = title:e_ge_castanea
		has_title = title:e_ge_waditic
		has_title = title:e_ge_tianzhao
		has_title = title:e_ge_hesthurgmani
		has_title = title:e_ge_bhajsingh
		has_title = title:e_ge_salbuthul
		has_title = title:e_ge_remant
		prestige_level >= 4
	}

	is_valid_showing_failures_only = {
		is_capable_adult = yes
		is_imprisoned = no
	}

	cost = {
		prestige = 2000
		piety = 1000
	}

	effect = {
		create_title_and_vassal_change = {
			type = created
			save_scope_as = change
			add_claim_on_loss = no
		}
		title:e_ge_new_earth = {
			change_title_holder = {
				holder = root
				change = scope:change
			}
		}
		resolve_title_and_vassal_change = scope:change
		add_prestige = 1000
		custom_tooltip = ge_form_new_earth_decision_effect_tt
	}

	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 100
	}
}
